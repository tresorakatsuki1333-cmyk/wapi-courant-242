Règles Firestore de Sécurité (Cloud)

Ces règles restreignent les écritures aux seuls utilisateurs vérifiés :

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function emailVerified() { return request.auth.token.email_verified == true; }
    function ownUid() { return request.resource.data.uid == request.auth.uid; }

    match /wapi_signals/{id} {
      allow read: if signedIn();
      allow create: if signedIn() && emailVerified() && ownUid();
    }
    match /wapi_votes/{uid} {
      allow read: if signedIn();
      allow create, update: if signedIn() && emailVerified() && uid == request.auth.uid;
    }
    match /{room}/{msg} {
      allow read: if signedIn() && room.matches('chat_[A-Za-z0-9_]+');
      allow create: if signedIn() && emailVerified() && room.matches('chat_[A-Za-z0-9_]+');
    }
  }
}
